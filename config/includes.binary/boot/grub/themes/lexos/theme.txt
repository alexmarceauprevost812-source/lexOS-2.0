#  ═══ LA ZONE TERMINAL EST PEINTE EN NOIR — C'ÉTAIT LE RECTANGLE BLEU ═══
#  ALEX, PHOTO DU ThinkPad : après avoir choisi LexOS, « un grand rectangle
#  bleu-turquoise à vagues » au milieu de l'écran, avec « Loading Linux… »
#  par-dessus. « On dirait que mon logiciel est complètement planté. »
#
#  Ce rectangle, c'est la zone terminal ci-dessus (8 % → 92 % en largeur,
#  10 % → 90 % en hauteur) : la fenêtre où GRUB écrit ses messages quand il
#  quitte le menu pour charger le noyau. Sans « terminal-box », GRUB n'y peint
#  rien à nous : il y laisse transparaître l'image de fond posée par
#  /etc/grub.d/05_debian_theme — celle de desktop-base, les vagues turquoise
#  de trixie. Le thème LexOS gagnait pour le MENU ; il ne disait rien de la
#  zone terminal, et c'est Debian qui la remplissait.
#
#  Neuf tranches (centre, quatre bords, quatre coins), toutes du noir plein
#  — sur le modèle des select_*.png. Avec ça, la zone est noire quel que soit
#  le fond en dessous ; /etc/default/grub.d/lexos.cfg coupe EN PLUS l'image
#  de desktop-base, pour que le repli n'ait jamais de bleu à montrer.
terminal-box:   "terminal_box_*.png"

# ───────────────────── LISTE DES DÉMARRAGES ─────────────────────
#  ═══ EN POURCENTAGES, ET C'EST TOUT LE CORRECTIF ═══
#  ALEX, ESSAI SUR L'ALIENWARE : « tout est déformé et les places pour le
#  boot ne sont pas bien placées ».
#
#  La géométrie était en PIXELS ABSOLUS, calée sur 1920x1080. Or le hook 0910
#  demande « gfxmode=1920x1080,1280x720,1024x768,auto » : dès que le firmware
#  ne rend pas le 1920x1080, GRUB prend un repli — et GRUB NE REDIMENSIONNE
#  PAS UN THÈME. Le fond, lui, est bien étiré à l'écran. Résultat : le cadre
#  dessiné se déplace, les entrées restent à 534/449, elles tombent à côté.
#  Et si le repli est du 1024x768 (4:3), un fond 16:9 étiré est déformé.
#  Les deux moitiés de la phrase d'Alex, et la même cause.
#
#  Un pourcentage, lui, suit l'étirement du fond exactement : le cadre est à
#  une FRACTION fixe de l'image, l'étirement l'envoie à la même fraction de
#  l'écran, et l'entrée posée à cette fraction-là reste dedans. Quelle que
#  soit la définition, quel que soit le rapport d'image.
#
#  MESURÉ DANS background.png, PAS LU DANS UN COMMENTAIRE (les ruptures de
#  luminance de l'image donnent le cadre au pixel près) :
#    cadre   : x 520 → 1400 (27,08 % → 72,92 %)   y 400 → 843 (37,04 % → 78,06 %)
#    entrées : x 534 → 1386 (27,81 % → 72,19 %)   y 449 (41,57 %), hauteur 380
#
#  Les pourcentages ci-dessous sont arrondis à l'entier VERS L'INTÉRIEUR du
#  cadre — GRUB fait « pourcentage x total / 100 » en entier, et une entrée
#  qui déborde d'un pixel se voit. Le banc tests/test_lexos_grub_theme.sh
#  refait ce calcul pour six définitions, dont du 4:3 et du 16:10.
#
#  CE QUI RESTE EN PIXELS, ET POURQUOI ON L'ASSUME : item_height et les
#  polices. La syntaxe des thèmes GRUB ne prend pas de pourcentage pour
#  celles-là. En basse définition, les entrées restent donc à leur taille et
#  la boîte rétrécit : le banc vérifie qu'il en tient au moins six partout,
#  parce qu'une entrée hors de la boîte est SILENCIEUSEMENT invisible
#  (scrollbar = false) — et choisir « secours » en croyant prendre « démo »,
#  c'est exactement ce qui met quelqu'un devant une console.
+ boot_menu {
    left   = 28%
    top    = 42%
    width  = 44%
    height = 35%

    #  52 ET NON 60, ET C'EST UNE MESURE, PAS UN GOÛT. La boîte vaut 35 % de
    #  la hauteur d'écran ; item_height, lui, ne sait pas être un pourcentage
    #  (la syntaxe des thèmes GRUB ne l'accepte pas). Calculé pour les six
    #  définitions du banc, avec 60 px : à 1280x720 il ne tenait plus que
    #  TROIS entrées sur les cinq du menu. On peut encore les atteindre — la
    #  liste défile à la flèche — mais voir trois entrées quand il y en a
    #  cinq, sur un écran où tout est déjà mal placé, c'est exactement la
    #  situation où l'on choisit « secours » en croyant prendre « démo ».
    #  52 px en rend quatre là, six à 1920x1080, et ne coûte que 13 % de
    #  hauteur de ligne à la définition de référence.
    item_height     = 52
    item_spacing    = 4
    item_padding    = 0
    item_icon_space = 18
    icon_width      = 46      # 22 px de marge transparente + 24 px de dessin
    icon_height     = 24

    item_font           = "LexOS Regular 22"
    selected_item_font  = "LexOS SemiBold 22"
    item_color          = "#c8c8cf"
    selected_item_color = "#ffffff"

    #  LES BANDES HAUTE ET BASSE DE select_*.png FONT 2 PIXELS, PAS 16.
    #  Ce n'est pas un choix esthétique. GRUB dessine le texte de l'entrée
    #  CHOISIE sous la bande haute du pixmap : avec 16 px, la première ligne
    #  descendait de 16 px et sa boîte orange mordait sur l'entrée suivante —
    #  l'écart 1->2 valait 48 px là où 2->3 en valait 64. Vérifié en machine
    #  virtuelle : ramener select_n/select_s (et les quatre coins) à 2 px de
    #  haut remet toutes les entrées à intervalle égal.
    #  Si un jour ces images sont redessinées : garder 2 px en haut et en bas.
    selected_item_pixmap_style = "select_*.png"

    scrollbar       = false
    scrollbar_width = 0
}
